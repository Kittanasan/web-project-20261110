/* ---------------- บันทึกผล + ตารางบันทึกผล ---------------- */
function confOf(level) {
  if (!currentData || !currentData.verdict) return null;
  if (level === currentData.verdict) return Number(currentData.confidence_pct || 0);
  const lp = currentData.level_pct || {};
  return lp[level] !== undefined ? Number(lp[level]) : null;
}

function buildRecord() {
  const modelLv = currentData && currentData.verdict ? currentData.verdict : null;
  const final = recResultEl.value || finalLevelEl.value || "Medium";
  const conf = confOf(final);
  const edited = !!modelLv && final !== modelLv;
  const label = (recLabelEl.value || sampleIdEl.value || "").trim();
  return {
    no: Number(recNoEl.value) || null,
    savedAt: new Date().toISOString(),
    label: label,
    sampleId: label,
    title: label || ((currentFile && currentFile.name) || "sample").replace(/\.[^.]+$/, ""),
    fileName: (currentFile && currentFile.name) || "",
    farm: (farm2El.value || farmEl.value || "").trim(),
    sampleName: "",
    inspector: (recInspectorEl.value || inspectorEl.value || "").trim(),
    note: noteEl.value.trim(),
    modelResult: modelLv,
    modelConfidence: currentData ? Number(currentData.confidence_pct || 0) : null,
    finalResult: final,
    finalConfidence: conf,
    overrideNote: finalNoteEl.value.trim(),
    edited: edited,
    detections: (currentData && currentData.detections) || [],
    imageDataUrl: currentAnnotated || originalDataUrl
  };
}

/* เติมค่าจากผลการประเมินลงในฟอร์มบันทึกผล */
function fillRecordForm(data) {
  if (!data || !data.verdict) { recConfEl.value = "—"; return; }
  recResultEl.value = data.verdict;
  const c = confOf(data.verdict);
  recConfEl.value = (c === null ? "" : c.toFixed(1)) + "%";
}

function pickCurrentResult() {
  if (!currentData || !currentData.verdict) {
    toast("ยังไม่มีผลจากแบบจำลอง — เลือกผลตรวจในช่อง ‘ผลตรวจที่ได้’ ได้เอง", true);
    recResultEl.focus();
    return;
  }
  recResultEl.value = currentData.verdict;
  finalLevelEl.value = currentData.verdict;
  fillRecordForm(currentData);
  syncRecordForm();
  checkOverride();
  finalNoteEl.value = "";
  toast("ใช้ผลจากแบบจำลอง: " + currentData.verdict);
}

function syncRecordForm() {
  const lv = recResultEl.value;
  const c = confOf(lv);
  recConfEl.value = c === null ? "" : c.toFixed(1) + "%";
  if (!recLabelEl.value.trim() && sampleIdEl.value.trim()) recLabelEl.value = sampleIdEl.value.trim();
  if (!farm2El.value.trim() && farmEl.value.trim()) farm2El.value = farmEl.value.trim();
  if (!recInspectorEl.value.trim() && inspectorEl.value.trim()) recInspectorEl.value = inspectorEl.value.trim();
}

/* ---------------- ที่เก็บข้อมูล (localStorage) ---------------- */
const HISTORY_KEY = "liverFatEvalRecordsV3";   // ตารางบันทึกผล
const IMAGE_KEY = "liverFatEvalImagesV1";      // ภาพผลประเมินแยกไว้ (กันพื้นที่หลักเต็ม)
const IMAGE_SAVE_LIMIT = 8;                    // เก็บภาพในเบราว์เซอร์สูงสุดกี่รายการ

function persistHistory() {
  const plain = history.map(h => {
    const c = Object.assign({}, h);
    delete c.imageDataUrl;
    return c;
  });
  try {
    localStorage.setItem(HISTORY_KEY, JSON.stringify(plain.slice(0, 300)));
  } catch (e) { /* พื้นที่เต็ม */ }

  /* เก็บภาพเฉพาะรายการใหม่สุด — ถ้าเต็มให้ลดจำนวนลง */
  try {
    const imgs = {};
    let n = 0;
    for (const h of history) {
      if (h.imageDataUrl && n < IMAGE_SAVE_LIMIT) { imgs[h.savedAt] = h.imageDataUrl; n++; }
    }
    localStorage.setItem(IMAGE_KEY, JSON.stringify(imgs));
  } catch (e) {
    try { localStorage.removeItem(IMAGE_KEY); } catch (e2) { /* ข้ามไป */ }
  }
}

function loadHistory() {
  try {
    const raw = localStorage.getItem(HISTORY_KEY);
    history = raw ? JSON.parse(raw) : [];
    if (!Array.isArray(history)) history = [];
  } catch (e) { history = []; }
  try {
    const rawImg = localStorage.getItem(IMAGE_KEY);
    const imgs = rawImg ? JSON.parse(rawImg) : {};
    history.forEach(h => { if (imgs[h.savedAt]) h.imageDataUrl = imgs[h.savedAt]; });
  } catch (e) { /* ไม่มีภาพเก็บไว้ */ }
  lastNo = history.reduce((m, h) => Math.max(m, Number(h.no) || 0), 0);
}

function fmtDate(iso) {
  const d = new Date(iso);
  if (isNaN(d.getTime())) return iso || "-";
  const p = (n) => String(n).padStart(2, "0");
  return p(d.getDate()) + "/" + p(d.getMonth() + 1) + "/" + (d.getFullYear() + 543) + " " + p(d.getHours()) + ":" + p(d.getMinutes());
}
function tagClass(lv) { return lv === "Hight" ? "t-h" : lv === "Medium" ? "t-m" : "t-l"; }
function levelText(lv) {
  if (lv === "Hight") return "ตรวจพบเม็ดไขมันมาก";
  if (lv === "Medium") return "ตรวจพบเม็ดไขมันปานกลาง";
  if (lv === "Low") return "ตรวจพบเม็ดไขมันน้อย";
  return "-";
}
function levelColor(lv) {
  if (lv === "Hight") return "#0e8fae";
  if (lv === "Medium") return "#c98a12";
  if (lv === "Low") return "#b3261e";
  return "#5b7288";
}
/* ---------------- วาดตารางบันทึกผล ---------------- */
function renderRecordTable(highlightTs) {
  /* รวมรายการที่บันทึกไว้ในเครื่อง (เซิร์ฟเวอร์) เข้ากับรายการในเบราว์เซอร์ โดยกันซ้ำด้วยชื่อไฟล์รายงาน */
  const seen = new Set(history.map(h => h.serverFile).filter(Boolean));
  let rows = history.slice();
  serverRecords.forEach(r => {
    if (r.report && seen.has(r.report)) return;
    rows.push({
      no: r.no, savedAt: r.saved_at || new Date((r.time || 0) * 1000).toISOString(),
      label: r.sample_name || r.sample_id || "-", sampleId: r.sample_id || "",
      farm: r.farm || "", inspector: r.inspector || "",
      finalResult: r.final_result || "", finalConfidence: r.final_confidence_pct,
      edited: !!r.edited, modelResult: "", modelConfidence: null,
      serverFile: r.report, serverImage: r.image, imageDataUrl: null, fromServer: true
    });
  });

  const q = (recordFilter.value || "").trim().toLowerCase();
  if (q) {
    rows = rows.filter(r => [r.label, r.sampleId, r.farm, r.inspector, r.finalResult]
      .join(" ").toLowerCase().indexOf(q) >= 0);
  }
  rows.sort((a, b) => (Number(a.no) || 0) - (Number(b.no) || 0));

  recordCount.textContent = rows.length + " รายการ";
  clearSavedBtn.disabled = history.length === 0 && serverRecords.length === 0;
  printBtn.disabled = rows.length === 0;
  recordEmpty.style.display = rows.length ? "none" : "block";

  recordBody.innerHTML = rows.map((r, i) => {
    const lv = r.finalResult;
    const conf = (r.finalConfidence === null || r.finalConfidence === undefined || r.finalConfidence === "")
      ? "—" : Number(r.finalConfidence).toFixed(1);
    const editedTag = r.edited ? '<div class="rec-num edited" style="font-size:.7rem;font-weight:600">* เจ้าหน้าที่แก้ไขผล</div>' : "";
    const isNew = highlightTs && r.savedAt === highlightTs ? " class=\"is-new\"" : "";
    return '<tr' + isNew + ' data-row="' + i + '">' +
      '<td class="rec-no">' + esc(r.no || (i + 1)) + "</td>" +
      '<td><b>' + esc(r.label || r.sampleId || "-") + "</b></td>" +
      '<td class="t-left">' + esc(r.farm || "-") + "</td>" +
      '<td><span class="rec-val" style="color:' + levelColor(lv) + ";border-color:" + levelColor(lv) +
        ";background:" + levelColor(lv) + '14">' + esc(lv || "-") +
        "<small>" + esc(levelText(lv)) + "</small></span></td>" +
      '<td class="rec-num">' + esc(conf) + (conf === "—" ? "" : " %") + "</td>" +
      '<td class="t-left">' + esc(r.inspector || "-") + editedTag + "</td>" +
      '<td style="font-size:.78rem;color:var(--muted)">' + esc(fmtDate(r.savedAt)) + "</td>" +
      '<td><div class="rec-actions">' +
        '<a class="act-view" href="#" data-view="' + i + '">ดูภาพ</a>' +
        '<a class="act-del" href="#" data-delete="' + i + '">ลบ</a>' +
      "</div></td></tr>";
  }).join("");
  renderRecordTable._rows = rows;
}

recordBody.addEventListener("click", async (e) => {
  const a = e.target.closest("a");
  if (!a) return;
  e.preventDefault();
  const rows = renderRecordTable._rows || [];
  if (a.dataset.view !== undefined) {
    const r = rows[Number(a.dataset.view)];
    if (!r) return;
    if (r.imageDataUrl) {
      showImageModal(r.imageDataUrl, "ผลประเมิน ลำดับที่ " + (r.no || "-") + " · " + (r.label || ""));
    } else if (r.serverImage && apiBase) {
      window.open(apiBase + "/exports/" + encodeURIComponent(r.serverImage), "_blank");
    } else {
      toast("ไม่พบภาพผลประเมินของรายการนี้", true);
    }
  } else if (a.dataset.delete !== undefined) {
    const r = rows[Number(a.dataset.delete)];
    if (!r) return;
    if (!confirm("ยืนยันการลบข้อมูลลำดับที่ " + (r.no || "-") + " (" + (r.label || "-") + ") ?" +
      (r.serverFile && apiBase ? "\nไฟล์ภาพและรายงานในโฟลเดอร์ exports จะถูกลบด้วย" : ""))) return;
    /* ลบในเบราว์เซอร์ */
    const idx = history.findIndex(h => h.savedAt === r.savedAt && (h.no || null) === (r.no || null));
    if (idx >= 0) history.splice(idx, 1);
    persistHistory();
    /* ลบไฟล์ฝั่งเซิร์ฟเวอร์ */
    if (r.serverFile && apiBase) {
      try {
        const res = await fetch(apiBase + "/delete", {
          method: "POST", headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ name: r.serverFile })
        });
        const j = await res.json();
        if (!res.ok || !j.ok) throw new Error(j.error || ("HTTP " + res.status));
        serverRecords = serverRecords.filter(s => s.report !== r.serverFile);
      } catch (err) {
        toast("ลบไฟล์ในโฟลเดอร์ exports ไม่สำเร็จ: " + err.message, true);
      }
    }
    renderRecordTable();
    setSaveStatus("ลบข้อมูลลำดับที่ " + (r.no || "-") + " แล้ว");
    toast("ลบข้อมูลแล้ว");
  }
});

/* ---------------- โหลดรายการที่บันทึกไว้ในเครื่อง ---------------- */
async function loadServerRecords(quiet) {
  if (!apiBase) return;
  try {
    const res = await fetch(apiBase + "/exports", { cache: "no-store" });
    const j = await res.json();
    if (!res.ok || !j.ok) throw new Error(j.error || ("HTTP " + res.status));
    serverRecords = j.exports || [];
    lastNo = Math.max(lastNo, ...serverRecords.map(s => Number(s.no) || 0));
    renderRecordTable();
    if (!quiet) toast("โหลดรายการจากโฟลเดอร์ exports แล้ว " + serverRecords.length + " รายการ");
  } catch (err) {
    if (!quiet) toast("โหลดรายการไม่สำเร็จ: " + err.message, true);
  }
}

function downloadJson(rows, name) {
  const clean = rows.map(r => {
    const c = Object.assign({}, r);
    delete c.imageDataUrl;
    return c;
  });
  const blob = new Blob([JSON.stringify(clean, null, 2)], { type: "application/json" });
  const link = document.createElement("a");
  link.href = URL.createObjectURL(blob);
  link.download = name;
  link.click();
  setTimeout(() => URL.revokeObjectURL(link.href), 4000);
}
